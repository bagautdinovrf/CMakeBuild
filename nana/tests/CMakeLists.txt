set(nana_test_targets controller_tests NanaUiSmoke NanaDpiTests NanaEditorTests NanaSafePlaceTests NanaUpdateBatchTests NanaMenuMetricsTests NanaUtf8Tests NanaBench)
# Optional paired measurement: the same menu fixture linked to a saved static
# Nana library from the baseline revision, without rebuilding that dependency.
set(CMAKEBUILD_NANA_BASELINE_LIBRARY "" CACHE FILEPATH "Saved Nana static library for menu benchmark comparison")
if(CMAKEBUILD_NANA_BASELINE_LIBRARY)
    add_executable(NanaMenuMetricsBaseline menu_metrics_tests.cpp "${PROJECT_BINARY_DIR}/app.rc")
    target_link_libraries(NanaMenuMetricsBaseline PRIVATE CMakeBuildCore "${CMAKEBUILD_NANA_BASELINE_LIBRARY}"
        "$<TARGET_PROPERTY:Nana,INTERFACE_LINK_LIBRARIES>")
    target_include_directories(NanaMenuMetricsBaseline SYSTEM PRIVATE "$<TARGET_PROPERTY:Nana,INTERFACE_INCLUDE_DIRECTORIES>")
    target_compile_definitions(NanaMenuMetricsBaseline PRIVATE "$<TARGET_PROPERTY:Nana,INTERFACE_COMPILE_DEFINITIONS>")
    list(APPEND nana_test_targets NanaMenuMetricsBaseline)
endif()
foreach(test_target IN LISTS nana_test_targets)
    if(MSVC)
        target_compile_options(${test_target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/std:c++latest;/W4;/permissive-;/utf-8;/Zc:__cplusplus>")
        target_link_options(${test_target} PRIVATE /MANIFEST:NO)
    elseif(MINGW)
        target_compile_options(${test_target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>")
        target_link_options(${test_target} PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
endforeach()
