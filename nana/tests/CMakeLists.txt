# Included by the variant's main project with CMAKEBUILD_BUILD_TESTS=ON.
# Link the same sources, compile definitions and static runtime as the panel.
add_executable(controller_tests controller_tests.cpp "${PROJECT_BINARY_DIR}/app.rc")
target_link_libraries(controller_tests PRIVATE CMakeBuildCore)
add_executable(NanaUiSmoke ui_smoke.cpp "${PROJECT_BINARY_DIR}/app.rc")
target_link_libraries(NanaUiSmoke PRIVATE CMakeBuildUi)
add_executable(NanaDpiTests dpi_tests.cpp "${PROJECT_BINARY_DIR}/app.rc")
target_link_libraries(NanaDpiTests PRIVATE CMakeBuildUi)
add_executable(NanaEditorTests editor_incremental_tests.cpp "${PROJECT_BINARY_DIR}/app.rc")
target_link_libraries(NanaEditorTests PRIVATE CMakeBuildCore Nana::Nana)
add_executable(NanaBench nana_bench.cpp "${PROJECT_BINARY_DIR}/app.rc")
target_link_libraries(NanaBench PRIVATE CMakeBuildUi psapi)
target_include_directories(NanaBench PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../benchmarks")
target_compile_definitions(NanaBench PRIVATE CMAKEBUILD_BENCH_APP_VERSION="${PROJECT_VERSION}")
target_compile_definitions(NanaBench PRIVATE CMAKEBUILD_BENCH_TOOLKIT_REVISION="${CMAKEBUILD_NANA_REVISION}")

foreach(test_target IN ITEMS controller_tests NanaUiSmoke NanaDpiTests NanaEditorTests NanaBench)
    if(MSVC)
        target_compile_options(${test_target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/utf-8;/Zc:__cplusplus>")
        target_link_options(${test_target} PRIVATE /MANIFEST:NO)
    elseif(MINGW)
        target_compile_options(${test_target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>")
        target_link_options(${test_target} PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
endforeach()

add_test(NAME nana_controller COMMAND controller_tests)
set_tests_properties(nana_controller PROPERTIES TIMEOUT 120 LABELS "integration;controller;processes")
add_test(NAME nana_ui COMMAND NanaUiSmoke "${CMAKE_CURRENT_BINARY_DIR}/ui-smoke")
set_tests_properties(nana_ui PROPERTIES TIMEOUT 45 LABELS "nana;ui")
add_test(NAME nana_dpi COMMAND NanaDpiTests "${CMAKE_CURRENT_BINARY_DIR}/dpi-tests")
set_tests_properties(nana_dpi PROPERTIES TIMEOUT 45 LABELS "nana;ui;dpi")
add_test(NAME nana_editor COMMAND NanaEditorTests)
set_tests_properties(nana_editor PROPERTIES TIMEOUT 45 LABELS "nana;ui;editor;regression")
