cmake_minimum_required(VERSION 3.24)
# Both panels share the root VERSION for their UI, PE metadata and manifest.
set(CMAKEBUILD_VERSION_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKEBUILD_VERSION_FILE}")
file(READ "${CMAKEBUILD_VERSION_FILE}" CMAKEBUILD_VERSION)
string(STRIP "${CMAKEBUILD_VERSION}" CMAKEBUILD_VERSION)
if(NOT CMAKEBUILD_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
    message(FATAL_ERROR "VERSION must contain MAJOR.MINOR.PATCH.")
endif()
foreach(component IN ITEMS "${CMAKE_MATCH_1}" "${CMAKE_MATCH_2}" "${CMAKE_MATCH_3}")
    if(component GREATER 65535)
        message(FATAL_ERROR "VERSION components must fit Windows version resources.")
    endif()
endforeach()
project(CMakeBuildNana VERSION "${CMAKEBUILD_VERSION}" LANGUAGES CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "The Nana variant currently supports Windows 10/11 x64.")
endif()
include(../cmake/cxx_standard.cmake)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(cmake/nana.cmake)
configure_file(src/version.hpp.in generated/version.hpp @ONLY)
configure_file(app.manifest.in app.manifest @ONLY)
file(TO_CMAKE_PATH "${CMAKE_CURRENT_BINARY_DIR}/app.manifest" APP_MANIFEST_PATH)
file(TO_CMAKE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/assets/app.ico" APP_ICON_PATH)
file(TO_CMAKE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/assets/Nana-LICENSE.txt" APP_LICENSE_PATH)
configure_file(app.rc.in app.rc @ONLY)
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/app.rc" PROPERTIES
    OBJECT_DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/app.manifest;${CMAKE_CURRENT_SOURCE_DIR}/assets/app.ico;${CMAKE_CURRENT_SOURCE_DIR}/assets/Nana-LICENSE.txt")
add_library(CMakeBuildCore STATIC src/engine.cpp src/platform.cpp src/app_state.cpp src/controller.cpp)
target_include_directories(CMakeBuildCore PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(CMakeBuildCore PUBLIC
    UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_link_libraries(CMakeBuildCore PUBLIC comdlg32 comctl32 shell32 ole32 uuid user32 gdi32 gdiplus dwmapi advapi32)
add_library(CMakeBuildUi STATIC src/ui.cpp src/settings_dialog.cpp)
target_link_libraries(CMakeBuildUi PUBLIC CMakeBuildCore Nana::Nana)
add_executable(CMakeBuild WIN32 src/main.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
target_link_libraries(CMakeBuild PRIVATE CMakeBuildUi)
foreach(application_target IN ITEMS CMakeBuildCore CMakeBuildUi CMakeBuild)
    if(MSVC)
        target_compile_options(${application_target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/std:c++latest;/W4;/permissive-;/utf-8;/Zc:__cplusplus>")
    elseif(MINGW)
        target_compile_options(${application_target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>")
    endif()
endforeach()
if(MSVC)
    target_link_options(CMakeBuild PRIVATE /MANIFEST:NO)
elseif(MINGW)
    target_link_options(CMakeBuild PRIVATE -municode -static -static-libgcc -static-libstdc++ "$<$<CONFIG:Release>:-s>")
endif()
set_target_properties(CMakeBuild PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin/$<CONFIG>")
set(CMAKEBUILD_EXECUTABLE "${CMAKE_CURRENT_SOURCE_DIR}/CMakeBuild.exe")
add_custom_command(TARGET CMakeBuild POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:CMakeBuild>" "${CMAKEBUILD_EXECUTABLE}" VERBATIM)
add_custom_target(CMakeBuildPublish ALL
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:CMakeBuild>" "${CMAKEBUILD_EXECUTABLE}" DEPENDS CMakeBuild VERBATIM)
install(TARGETS CMakeBuild RUNTIME DESTINATION .)
option(CMAKEBUILD_BUILD_TESTS "Build isolated controller and Nana widget checks" OFF)
if(CMAKEBUILD_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
